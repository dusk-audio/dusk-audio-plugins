cmake_minimum_required(VERSION 3.15)
project(ChordAnalyzerDAF VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

include("${CMAKE_CURRENT_SOURCE_DIR}/../../shared-daf/cmake/DuskDafPlugin.cmake")
enable_testing()
add_subdirectory(../core core)

set(CHORD_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
dusk_daf_embed_font(kChordFontRegular "${DUSK_SHARED_DAF_DIR}/fonts/BarlowCondensed/BarlowCondensed-Regular.ttf" "${CHORD_GENERATED_DIR}/ChordFontRegular.inc")
dusk_daf_embed_font(kChordFontSemiBold "${DUSK_SHARED_DAF_DIR}/fonts/BarlowCondensed/BarlowCondensed-SemiBold.ttf" "${CHORD_GENERATED_DIR}/ChordFontSemiBold.inc")
daf_add_plugin(chord-analyzer-2 TARGETS clap vst3 lv2 au MONOLITHIC USE_FILE_BROWSER UI_TYPE opengl
    FILES_DSP ChordAnalyzerPlugin.cpp FILES_UI ChordAnalyzerUI.cpp ${DUSK_DAF_UI_SOURCES})
daf_add_plugin(chord-analyzer-midi-2 TARGETS clap vst3 lv2 au MONOLITHIC USE_FILE_BROWSER UI_TYPE opengl
    FILES_DSP ChordAnalyzerPlugin.cpp FILES_UI ChordAnalyzerUI.cpp ${DUSK_DAF_UI_SOURCES})
foreach(plugin chord-analyzer-2 chord-analyzer-midi-2)
    if(plugin STREQUAL "chord-analyzer-midi-2")
        set(mode 1)
    else()
        set(mode 0)
    endif()
    target_compile_definitions(${plugin} PUBLIC CHORD_ANALYZER_DAF_MIDI_MODE=${mode}
        CA2_VERSION_MAJOR=${PROJECT_VERSION_MAJOR} CA2_VERSION_MINOR=${PROJECT_VERSION_MINOR}
        CA2_VERSION_PATCH=${PROJECT_VERSION_PATCH})
    target_include_directories(${plugin} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CHORD_GENERATED_DIR}" ${DUSK_DAF_INCLUDE_DIRS})
    target_link_libraries(${plugin} PUBLIC ChordAnalyzerRuntime ChordMidiExport)
    dusk_daf_install_local(${plugin})

    if(NOT CMAKE_CROSSCOMPILING)
        add_executable(${plugin}-test tests/ChordAnalyzerPluginTest.cpp
            "${DAF_PATH}/daf/src/DafPlugin.cpp")
        target_link_libraries(${plugin}-test PRIVATE ${plugin})
        target_compile_definitions(${plugin}-test PRIVATE DAF_PLUGIN_TARGET_CLAP=1)
        add_test(NAME ${plugin} COMMAND ${plugin}-test)
        set_tests_properties(${plugin} PROPERTIES TIMEOUT 30)
        if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            target_compile_definitions(${plugin}-test PRIVATE CHORD_TEST_WRAP_MALLOC=1)
            target_compile_options(${plugin}-test PRIVATE -fno-builtin-malloc -fno-builtin-calloc -fno-builtin-realloc)
            target_link_options(${plugin}-test PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
        endif()
    endif()
endforeach()

if(NOT CMAKE_CROSSCOMPILING)
    add_executable(ChordAnalyzerDafLv2Test tests/ChordAnalyzerLv2Test.cpp)
    target_include_directories(ChordAnalyzerDafLv2Test PRIVATE "${DAF_PATH}/daf/src")
    target_link_libraries(ChordAnalyzerDafLv2Test PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    foreach(plugin chord-analyzer-2 chord-analyzer-midi-2)
        if(plugin STREQUAL "chord-analyzer-midi-2")
            set(mode 1)
        else()
            set(mode 0)
        endif()
        add_dependencies(ChordAnalyzerDafLv2Test ${plugin}-lv2)
        add_test(NAME ${plugin}-lv2 COMMAND ChordAnalyzerDafLv2Test $<TARGET_FILE:${plugin}-lv2> ${mode})
        set_tests_properties(${plugin}-lv2 PROPERTIES TIMEOUT 10)
    endforeach()
endif()

if(NOT CMAKE_CROSSCOMPILING AND NOT WIN32 AND NOT APPLE)
    find_package(X11 REQUIRED)
    add_executable(ChordAnalyzerUiTest tests/ChordAnalyzerUiTest.cpp)
    target_include_directories(ChordAnalyzerUiTest PRIVATE "${DAF_PATH}/daf/src")
    target_link_libraries(ChordAnalyzerUiTest PRIVATE ChordMidiExport Threads::Threads ${CMAKE_DL_LIBS} ${X11_X11_LIB} ${X11_XTest_LIB})
    foreach(plugin chord-analyzer-2 chord-analyzer-midi-2)
        add_dependencies(ChordAnalyzerUiTest ${plugin}-clap)
        add_test(NAME ${plugin}Ui COMMAND ChordAnalyzerUiTest "$<TARGET_FILE:${plugin}-clap>" "${CMAKE_CURRENT_BINARY_DIR}/ui-artifacts/${plugin}")
        set_tests_properties(${plugin}Ui PROPERTIES TIMEOUT 90)
        dusk_daf_add_resize_test(${plugin})
        set_tests_properties(${plugin}Ui ${plugin}Resize PROPERTIES RUN_SERIAL TRUE)
    endforeach()
endif()

# Retain the shipping headless URI/19-port ABI alongside the desktop successors.
# It is a native numeric-core LV2, with no JUCE, DAF UI or display dependency.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_path(CHORD_ANALYZER_LV2_HEADERS lv2/core/lv2.h
        DOC "Directory containing the public LV2 SDK's lv2/ headers")
    if(NOT CHORD_ANALYZER_LV2_HEADERS)
        message(FATAL_ERROR "Install the LV2 SDK or set CHORD_ANALYZER_LV2_HEADERS")
    endif()
    include(../lv2/ChordAnalyzerHeadless.cmake)
    chord_add_headless_lv2("${CMAKE_CURRENT_SOURCE_DIR}/../lv2" "${CHORD_ANALYZER_LV2_HEADERS}"
        "${CMAKE_CURRENT_BINARY_DIR}/bin/Chord Analyzer Headless.lv2")
endif()
